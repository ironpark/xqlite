# UTF-16 databases (PRAGMA encoding before the database is made). Text
# is stored and compared as UTF-16: BINARY compares the stored bytes, so
# the order of an index differs from UTF-8 (U+0100 sorts before 'a' in
# UTF-16le); NOCASE and RTRIM are defined for UTF-8 only and compare
# converted copies. Functions read their arguments as UTF-8 except
# hex(), octet_length() and the blob forms.

CREATE INDEX te ON t(lower(b));
