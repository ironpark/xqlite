-- xqt wire-diff pg: what a psql user sees, against PostgreSQL 16 and
-- `xqlite serve --pg` (tools/xqt/wire_diff.zig). The lines before the
-- first case are the setup of every case (run quietly); "-- case: NAME"
-- starts a case. psql echoes each case's input.
CREATE TABLE author (id serial PRIMARY KEY, name text NOT NULL UNIQUE, born date);
CREATE TABLE book (
  id integer PRIMARY KEY,
  author_id integer NOT NULL REFERENCES author (id) ON DELETE CASCADE,
  title varchar(200) NOT NULL,
  price numeric(10,2) DEFAULT 0 CHECK (price >= 0),
  tags jsonb,
  published boolean DEFAULT false
);
CREATE INDEX book_title ON book (title);
CREATE UNIQUE INDEX book_author_title ON book (author_id, title);
CREATE VIEW cheap_book AS SELECT id, title, price FROM book WHERE price < 10;
CREATE SEQUENCE ticket START 100;
INSERT INTO author (name, born) VALUES ('Ann', '1960-01-02'), ('Bob', NULL);
INSERT INTO book VALUES (1, 1, 'Alpha', 9.50, '{"a": 1}', true), (2, 1, 'Beta', 12, NULL, false), (3, 2, 'Gamma', NULL, '[1, 2]', NULL);

-- case: info-schema-constraints
SELECT constraint_name, constraint_type FROM information_schema.table_constraints WHERE table_name = 'book' AND constraint_type <> 'CHECK' ORDER BY constraint_name;
SELECT kcu.column_name, ccu.table_name AS foreign_table, ccu.column_name AS foreign_column
FROM information_schema.key_column_usage kcu
JOIN information_schema.constraint_column_usage ccu ON ccu.constraint_name = kcu.constraint_name
JOIN information_schema.table_constraints tc ON tc.constraint_name = kcu.constraint_name
WHERE tc.constraint_type = 'FOREIGN KEY' AND kcu.table_name = 'book';
