-- Differential cases of PostgreSQL's uuid (xqt dialect-diff pg
-- tools/dialect/pg-uuid-cases.txt), in the format of pg-cases.txt: each
-- case starts with a "-- case:" line and runs on an empty database, in
-- PostgreSQL (psql, unaligned, with headers) and in `xqlite sql` with
-- `.dialect postgres`; the rows, column names and error messages (the
-- "ERROR:" line) must be the same. The expected output is PostgreSQL 18's.
-- Random and time-based uuids are seen through what is certain of them:
-- their version, variant, length, order and count.

-- case: input forms uuid_in() reads, written lower case with hyphens
SELECT 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11'::uuid AS a, 'A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11'::uuid AS b, '{a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11}'::uuid AS c;
SELECT 'a0eebc999c0b4ef8bb6d6bb9bd380a11'::uuid AS d, 'a0ee-bc99-9c0b-4ef8-bb6d-6bb9-bd38-0a11'::uuid AS e, '{a0eebc999c0b4ef8bb6d6bb9bd380a11}'::uuid AS f;
SELECT 'a0eebc99-9c0b4ef8-bb6d6bb9-bd380a11'::uuid AS g, CAST('00000000-0000-0000-0000-000000000000' AS uuid) AS h, uuid 'ffffffff-ffff-ffff-ffff-ffffffffffff' AS i;
SELECT NULL::uuid IS NULL AS n;

-- case: input errors
SELECT 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a1'::uuid;
SELECT 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11-'::uuid;
SELECT 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a111'::uuid;
SELECT ' a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11'::uuid;
SELECT 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11 '::uuid;
SELECT '{a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11'::uuid;
SELECT 'a0e-ebc99-9c0b-4ef8-bb6d-6bb9bd380a11'::uuid;
SELECT 'a0eebc99--9c0b-4ef8-bb6d-6bb9bd380a11'::uuid;
SELECT 'g0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11'::uuid;
SELECT ''::uuid;

-- case: a uuid column: storage, order, comparisons with literals
CREATE TABLE t (id int, u uuid PRIMARY KEY);
INSERT INTO t VALUES (1, '{11111111-1111-1111-1111-111111111111}'), (2, '22222222222222222222222222222222'), (3, 'FFFFFFFF-0000-0000-0000-000000000000'), (4, '3f3e3c3b-3a30-3938-3736-353433a2313e');
SELECT id, u FROM t ORDER BY u;
SELECT id, u FROM t ORDER BY u DESC;
SELECT id FROM t WHERE u = '22222222-2222-2222-2222-222222222222';
SELECT id FROM t WHERE u = 'FFFFFFFF000000000000000000000000';
SELECT count(*) FROM t WHERE u <> '11111111111111111111111111111111';
SELECT count(*) FROM t WHERE u < '3f3e3c3b-3a30-3938-3736-353433a2313e';
SELECT count(*) FROM t WHERE u >= '3f3e3c3b-3a30-3938-3736-353433a2313e';
SELECT id FROM t WHERE u IN ('22222222-2222-2222-2222-222222222222', '{ffffffff-0000-0000-0000-000000000000}') ORDER BY id;
SELECT id FROM t WHERE u NOT IN ('22222222-2222-2222-2222-222222222222') ORDER BY id;
SELECT id FROM t WHERE u BETWEEN '20000000-0000-0000-0000-000000000000' AND '40000000-0000-0000-0000-000000000000' ORDER BY id;
SELECT id FROM t WHERE u = '22222222-2222-2222-2222-222222222222'::uuid;
SELECT min(id) FROM t GROUP BY u > '30000000-0000-0000-0000-000000000000' ORDER BY 1;

-- case: storing bad values, duplicates
CREATE TABLE t (u uuid UNIQUE);
INSERT INTO t VALUES ('11111111-1111-1111-1111-111111111111');
INSERT INTO t VALUES ('not a uuid');
INSERT INTO t VALUES ('11111111111111111111111111111111');
UPDATE t SET u = 'x';
UPDATE t SET u = '{22222222-2222-2222-2222-222222222222}';
SELECT u FROM t;

-- case: defaults and keys
CREATE TABLE p (id uuid PRIMARY KEY DEFAULT gen_random_uuid(), k uuid DEFAULT 'A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11', v uuid DEFAULT uuidv7(), name text);
INSERT INTO p (name) VALUES ('a'), ('b');
SELECT name, uuid_extract_version(id) AS vid, k, uuid_extract_version(v) AS vv FROM p ORDER BY name;
SELECT count(DISTINCT id) AS ids, count(DISTINCT k) AS ks FROM p;
CREATE TABLE q (pid uuid REFERENCES p, n int);
INSERT INTO q SELECT id, 1 FROM p WHERE name = 'a';
INSERT INTO q VALUES ('00000000-0000-0000-0000-000000000000', 2);
SELECT p.name, q.n FROM p JOIN q ON q.pid = p.id;

-- case: casts and text
CREATE TABLE t (u uuid);
INSERT INTO t VALUES ('a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11');
SELECT u::text AS t, CAST(u AS varchar) AS v, length(u::text) AS n, u::text = 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11' AS eq FROM t;
SELECT u || '!' AS a, 'id:' || u AS b, concat(u, '/', u) AS c, format('%s', u) AS d FROM t;
SELECT upper(u::text) AS up, u::text::uuid = u AS round FROM t;
SELECT 'A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11'::uuid::text AS lit;

-- case: uuid in JSON and arrays
CREATE TABLE t (id int, u uuid);
INSERT INTO t VALUES (1, 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11'), (2, '11111111-1111-1111-1111-111111111111');
SELECT to_json(u) AS j, to_jsonb(u) AS jb, json_build_object('u', u) AS o FROM t ORDER BY id;
SELECT json_agg(u ORDER BY id) AS j FROM t;
SELECT ARRAY[u] AS a FROM t ORDER BY id;
SELECT array_agg(u ORDER BY u) AS a FROM t;
SELECT id FROM t WHERE u = ANY ('{11111111-1111-1111-1111-111111111111,22222222-2222-2222-2222-222222222222}'::uuid[]);
SELECT id FROM t WHERE u = ANY (ARRAY['A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11'::uuid]);
SELECT '{A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11,{11111111111111111111111111111111}}'::uuid[] AS a;

-- case: a uuid[] column
CREATE TABLE t (id int, us uuid[]);
INSERT INTO t VALUES (1, '{A0EEBC99-9C0B-4EF8-BB6D-6BB9BD380A11,NULL}'), (2, ARRAY['11111111-1111-1111-1111-111111111111'::uuid]);
SELECT id, us, cardinality(us) AS n FROM t ORDER BY id;
SELECT id, us[1] AS first FROM t ORDER BY id;
INSERT INTO t VALUES (3, '{bad}');

-- case: generated uuids: version, variant, length
SELECT uuid_extract_version(gen_random_uuid()) AS g, uuid_extract_version(uuidv4()) AS v4, uuid_extract_version(uuidv7()) AS v7;
SELECT length(gen_random_uuid()::text) AS n, substr(uuidv4()::text, 15, 1) AS ver4, substr(uuidv7()::text, 15, 1) AS ver7;
SELECT substr(gen_random_uuid()::text, 20, 1) IN ('8', '9', 'a', 'b') AS variant4, substr(uuidv7()::text, 20, 1) IN ('8', '9', 'a', 'b') AS variant7;
SELECT gen_random_uuid() <> gen_random_uuid() AS differ, uuidv4() <> uuidv4() AS differ4;

-- case: uuidv7 is ordered within the session
CREATE TABLE t (id serial, u uuid);
INSERT INTO t (u) SELECT uuidv7() FROM generate_series(1, 20);
SELECT array_agg(id ORDER BY u) AS ids FROM t;
SELECT count(DISTINCT u) AS n FROM t;
INSERT INTO t (u) VALUES (uuidv7());
INSERT INTO t (u) VALUES (uuidv7());
SELECT max(id) = 22 AS last FROM t WHERE u = (SELECT u FROM t ORDER BY u DESC LIMIT 1);

-- case: uuidv7 with a shift
SELECT uuid_extract_version(uuidv7(interval '1 day')) AS v, uuidv7(interval '1 day') > uuidv7() AS later, uuidv7(interval '-1 day') < uuidv7() AS earlier;
SELECT uuid_extract_timestamp(uuidv7(interval '1000 years')) > now() + interval '999 years' AS far;
SELECT uuidv7('-1000 years'::interval);
SELECT uuidv7('9000 years'::interval);
SELECT uuidv7(NULL) IS NULL AS n;

-- case: uuid_extract_version and uuid_extract_timestamp
SELECT uuid_extract_version('11111111-1111-5111-8111-111111111111') AS v5, uuid_extract_version('11111111-1111-1111-1111-111111111111') AS bad_variant, uuid_extract_version('11111111-1111-f111-b111-111111111111') AS v15;
SELECT uuid_extract_version(NULL) AS n, uuid_extract_version('00000000-0000-0000-0000-000000000000') AS nil;
SELECT uuid_extract_timestamp('C232AB00-9414-11EC-B3C8-9F6BDECED846') AS v1, uuid_extract_timestamp('017F22E2-79B0-7CC3-98C4-DC0C0C07398F') AS v7;
SELECT uuid_extract_timestamp('017f22e2-79b0-7cc3-98c4-dc0c0c07398f') = '2022-02-22 19:22:22+00'::timestamptz AS v7_eq, uuid_extract_timestamp('C232AB00-9414-11EC-B3C8-9F6BDECED846') = '2022-02-22 14:22:22-05'::timestamptz AS v1_eq;
SELECT uuid_extract_timestamp('11111111-1111-1111-1111-111111111111') AS nul, uuid_extract_timestamp('11111111-1111-4111-8111-111111111111') AS v4;
SELECT uuid_extract_timestamp('00000000-0000-1000-8000-000000000000') AS gregorian, uuid_extract_timestamp('00000000-0000-7000-8000-000000000000') AS epoch;
SELECT uuid_extract_timestamp(uuidv7()) <= now() + interval '1 minute' AS now7;
