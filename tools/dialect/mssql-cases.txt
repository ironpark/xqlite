-- Differential cases for the T-SQL dialect (xqt dialect-diff mssql).
-- Each case starts with a "-- case:" line and runs on an empty database
-- called main, in SQL Server (sqlcmd -W -s '|', SET NOCOUNT ON) and in
-- `xqlite sql` with `.dialect mssql`; the rows, column names and error
-- messages (the "ERROR:" line) must be the same. GO ends a batch: a
-- statement that fails to compile in SQL Server skips the rest of its
-- batch, so such statements end with GO. What the two print differently
-- by design (docs/mssql.md: DECIMAL as REAL, the engine's own messages,
-- GETDATE()) is avoided; numbers with a fraction are compared without
-- trailing zeros.

-- case: arithmetic and operators
SELECT 1 + 1 AS a, 7 / 2 AS b, -7 / 2 AS c, 7 % 3 AS d, 7.0 / 2 AS e, 2 * 3 + 4 AS f;
SELECT 6 & 3 AS a, 6 | 1 AS b, 5 ^ 3 AS c, ~0 AS d, 10 - 2 - 3 AS e, (2 + 3) * 4 AS f;
SELECT 'a' + 'b' AS a, N'x' + 'y' + 'z' AS b, 'abcd' + NULL AS c, '1' + 1 AS d, 1 + '2' AS e;
GO

-- case: comparisons and predicates
SELECT CASE WHEN 'abc' = 'ABC' THEN 1 ELSE 0 END AS ci, CASE WHEN 'abc' LIKE 'A%' THEN 1 ELSE 0 END AS l;
SELECT CASE WHEN 2 BETWEEN 1 AND 3 THEN 1 ELSE 0 END AS a, CASE WHEN 5 NOT BETWEEN 1 AND 3 THEN 1 ELSE 0 END AS b,
  CASE WHEN 2 IN (1, 2) THEN 1 ELSE 0 END AS c, CASE WHEN 2 NOT IN (1, 3) THEN 1 ELSE 0 END AS d,
  CASE WHEN NULL IS NULL THEN 1 ELSE 0 END AS e, CASE WHEN 1 !< 2 THEN 1 ELSE 0 END AS f, CASE WHEN 1 !> 2 THEN 1 ELSE 0 END AS g;
SELECT CASE WHEN 'a_c' LIKE 'a[_]c' THEN 1 ELSE 0 END AS a, CASE WHEN 'abc' LIKE 'a[_]c' THEN 1 ELSE 0 END AS b,
  CASE WHEN 'bcd' LIKE '[a-c]%' THEN 1 ELSE 0 END AS c, CASE WHEN 'dcd' LIKE '[^a-c]%' THEN 1 ELSE 0 END AS d,
  CASE WHEN '50%' LIKE '50!%' ESCAPE '!' THEN 1 ELSE 0 END AS e, CASE WHEN 'ABC' LIKE '%b%' THEN 1 ELSE 0 END AS f;
SELECT CASE 2 WHEN 1 THEN 'a' WHEN 2 THEN 'b' END AS x, CASE WHEN 1 = 0 THEN 'yyyy' END AS y;
GO

CREATE TABLE a (id INT PRIMARY KEY, x VARCHAR(5));

INSERT INTO a VALUES (1, 'p'), (2, 'q'), (3, 'r');
INSERT INTO b VALUES (10, 1, 'u'), (11, 1, 'v'), (12, 2, 'w'), (13, 9, 'z');
SELECT a.x, b.y FROM a INNER JOIN b ON b.a_id = a.id ORDER BY b.y;
SELECT a.x, b.y FROM a LEFT OUTER JOIN b ON b.a_id = a.id ORDER BY a.x, b.y;
SELECT a.x, b.y FROM a RIGHT JOIN b ON b.a_id = a.id ORDER BY b.y;
SELECT a.x, b.y FROM a FULL OUTER JOIN b ON b.a_id = a.id ORDER BY a.x, b.y;
SELECT COUNT(*) AS n FROM a CROSS JOIN b;
SELECT x FROM a WHERE EXISTS (SELECT 1 FROM b WHERE b.a_id = a.id) ORDER BY x;
SELECT x, (SELECT COUNT(*) FROM b WHERE b.a_id = a.id) AS n FROM a ORDER BY x;
SELECT x FROM a WHERE id IN (SELECT a_id FROM b) ORDER BY x;
SELECT d.x, d.n FROM (SELECT x, id * 2 AS n FROM a) AS d WHERE d.n > 2 ORDER BY d.n;
SELECT v.p, v.q FROM (VALUES (1, 'one'), (2, 'two')) AS v(p, q) ORDER BY v.p;

-- case: update and delete

INSERT INTO c VALUES (1, 'one', 2), (2, 'two', 3);
UPDATE p SET price = price + 1 WHERE id = 1;
UPDATE p SET price *= 2 OUTPUT INSERTED.id, INSERTED.price WHERE id = 2;
UPDATE p SET p.price = p.price * c.rate FROM p JOIN c ON c.id = p.cat;
SELECT id, price FROM p ORDER BY id;
UPDATE x SET x.name = c.label FROM p AS x INNER JOIN c ON c.id = x.cat WHERE x.id > 1;
SELECT id, name FROM p ORDER BY id;
UPDATE TOP (1) p SET name = 'top' WHERE cat = 3;
DELETE FROM p OUTPUT DELETED.id WHERE id = 4;
DELETE p FROM p JOIN c ON c.id = p.cat WHERE c.label = 'one';

-- case: merge
CREATE TABLE tgt (id INT PRIMARY KEY, v VARCHAR(10), n INT);
INSERT INTO tgt VALUES (1, 'old1', 1), (2, 'old2', 2), (3, 'old3', 3);
MERGE INTO tgt AS t
USING (VALUES (2, 'new2'), (3, 'del3'), (4, 'new4')) AS s (id, v)
ON t.id = s.id
WHEN MATCHED AND s.v LIKE 'del%' THEN DELETE
WHEN MATCHED THEN UPDATE SET v = s.v, n = t.n + 10
WHEN NOT MATCHED BY TARGET THEN INSERT (id, v, n) VALUES (s.id, s.v, 0)
WHEN NOT MATCHED BY SOURCE THEN UPDATE SET v = 'gone';
SELECT id, v, n FROM tgt ORDER BY id;
MERGE tgt USING (SELECT 1 AS id) AS s ON tgt.id = s.id WHEN MATCHED THEN DELETE;
SELECT COUNT(*) AS n FROM tgt;
MERGE tgt AS t USING (VALUES (5, 'five'), (2, 'two')) AS s (id, v) ON t.id = s.id
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.v, 5)
WHEN MATCHED THEN DELETE;
SELECT id, v, n FROM tgt ORDER BY id;
GO

-- case: merge upsert
CREATE TABLE stock (sku VARCHAR(10) PRIMARY KEY, qty INT NOT NULL);
INSERT INTO stock VALUES ('a', 1), ('b', 2);
CREATE TABLE incoming (sku VARCHAR(10), qty INT);
INSERT INTO incoming VALUES ('b', 10), ('c', 5);
MERGE stock AS t USING incoming AS s ON t.sku = s.sku
WHEN MATCHED THEN UPDATE SET t.qty = t.qty + s.qty
WHEN NOT MATCHED THEN INSERT (sku, qty) VALUES (s.sku, s.qty);
SELECT sku, qty FROM stock ORDER BY sku;
MERGE stock AS t USING (SELECT sku FROM incoming) AS s ON t.sku = s.sku
WHEN NOT MATCHED BY SOURCE THEN DELETE;
SELECT sku, qty FROM stock ORDER BY sku;
GO
