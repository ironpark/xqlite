-- Differential cases of double precision and real (xqt dialect-diff pg
-- tools/dialect/pg-float-cases.txt), in the format of pg-cases.txt: their
-- input (float8in(), float4in()), a real rounded to single precision,
-- their text (float8out()), -0, and numbers compared with string
-- literals. NaN is not one (a SQLite value cannot be NaN).

-- case: input of strings
SELECT 'Infinity'::float8 AS a, '-inf'::float8 AS b, ' 1.5 '::float8 AS c, '+INFINITY'::real AS d, float8 '2e3' AS e;
SELECT 'abc'::float8;
SELECT '1e400'::float8;
SELECT '1e40'::real;
SELECT '1e-50'::real;
SELECT '1e-400'::float8;
SELECT (1e300::numeric)::real;
SELECT (1e-60::numeric)::real;
SELECT 'Infinity'::numeric::float8 AS a, '-inf'::numeric::real AS b, 1.25::numeric::float8 AS c, float4(0.1) = 0.1::float4 AS d, float8('2.5') AS e;
SELECT x::float8 AS f, x::real AS r FROM (VALUES ('2.5'), ('-0'), ('1e10')) v(x);

-- case: real is single precision
SELECT 1.1::real::float8 AS a, '1.1'::real::float8 AS b, (2^24 + 1)::bigint::real::float8 AS c, 3::real AS d, pg_typeof(1.5::real) AS t;
SELECT 1e300::float8::real;
SELECT 1e-300::float8::real;
CREATE TABLE fr (id int, a float8, b real);
INSERT INTO fr VALUES (1, '1.1', '1.1'), (2, 'Infinity', '-inf'), (3, 1.1, 1.1), (4, 2, 1e10::float8);
SELECT id, a, b, b::float8 AS bd, a::text AS at, b::text AS bt FROM fr ORDER BY id;
INSERT INTO fr VALUES (5, 'abc', 1);
INSERT INTO fr (id, b) VALUES (6, 1e300::float8);
SELECT 999999::real AS a, 1000000::real AS b, 1234567::real AS c, 123456.7::real AS d, 0.00001234::real AS e;

-- case: text of doubles
SELECT 1e10::float8::text AS a, 2.0::float8::text AS b, 1e15::float8::text AS c, 1e-5::float8::text AS d, 0.1::real::text AS e, 'Infinity'::float8::text AS f, (0.1::float8 + 0.2::float8)::text AS g;
SELECT 'a' || 1.5::float8 || 'b' AS c, concat(1e20::float8, '/', 0.1::real) AS cc, format('%s', 1e10::float8) AS f;

-- case: unary minus of doubles and Infinity
SELECT -0.0::float8 AS a, -(0.0::float8) AS b, -(-(0.0::float8)) AS c, -'Infinity'::float8 AS d, -(1.5::real) AS e, pg_typeof(-(1.5::real)) AS t;
SELECT 'Infinity'::float8 + 1 AS a, -'Infinity'::float8 < 0 AS b, 'Infinity'::float8 > 1e308 AS c;

-- case: a number compared with a string literal
SELECT 10 > '7' AS a, 10::int > '7' AS b, 2.5::float8 > '10' AS c, 2::int2 = '2' AS d, '9' < 10::bigint AS e;
SELECT x, x > '7' AS gt FROM (VALUES (5), (10)) v(x) ORDER BY x;
SELECT 1 = 'a';

-- case: numerics of different scales in a compound's column
SELECT x FROM (VALUES (2.50), (1.0), (3)) v(x) ORDER BY x;
SELECT x FROM (SELECT 2.50 AS x UNION ALL SELECT 1.0) v ORDER BY x DESC;
SELECT x, x * 2 AS d FROM (VALUES (2.50), (-1.0)) v(x) ORDER BY x;
