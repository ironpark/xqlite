-- Differential cases for the PostgreSQL dialect (xqt dialect-diff pg).
-- Each case starts with a "-- case:" line and runs on an empty database,
-- in PostgreSQL (psql, unaligned, with headers) and in `xqlite sql` with
-- `.dialect postgres`; the rows, column names and error messages (the
-- "ERROR:" line) must be the same. Booleans are cast to int and floats
-- avoided where the two print them differently.

-- case: names fold to lower case, quoted ones keep it
CREATE TABLE Mixed (Id int, "Camel" text);
INSERT INTO MIXED VALUES (1, 'a');
SELECT ID, "Camel" FROM mixed;
SELECT tablename FROM pg_tables WHERE schemaname = 'public';

SELECT (SELECT b FROM t), EXISTS (SELECT 1)::int, 'lit', 2 AS two;

SELECT id FROM n ORDER BY id OFFSET 1 ROWS FETCH FIRST 2 ROWS ONLY;

SELECT 'pg_class'::regclass::oid::int, 'int4'::regtype::oid::int, 'text'::regtype, 'pg_catalog'::regnamespace::oid::int;

-- case: date, timestamp and time operators
SELECT date '2024-01-01' + 1 AS a, 1 + date '2024-12-31' AS b, date '2024-03-01' - 1 AS c, date '2024-03-01' - date '2024-02-01' AS d, date '2024-01-02' - '2024-01-01' AS e;
SELECT date '2024-01-31' + interval '1 month' AS a, date '2023-01-31' + interval '1 month' AS b, date '2024-01-01' - interval '1 hour' AS c, date '2024-01-01' + time '10:30' AS d;
SELECT timestamp '2024-01-31 10:00' + interval '1 month 1 day' AS a, timestamp '2024-01-01' + '1 day' AS b, timestamp '2024-01-02 02:00' - timestamp '2024-01-01' AS c, timestamp '2024-01-01' - timestamp '2024-01-02 02:00' AS d;
SELECT timestamp '2024-01-01 10:00' - date '2024-01-01' AS a, timestamp '2024-01-02 03:04:05.5' AS b, time '23:00' + interval '2 hours' AS c, time '10:00' - time '08:30' AS d;

-- case: interval operators and output
SELECT interval '1 day' AS a, interval '1 day 2 hours' AS b, interval '38 months' AS c, interval '90 minutes' AS d, -interval '1 month' AS e, interval '-1 day' + interval '1 hour' AS f;
SELECT interval '1 year 2 months' + interval '1 year' AS a, interval '1 day' - interval '2 hours' AS b, interval '1 day' * 2 AS c, 2 * interval '1 hour' AS d, interval '1 month' / 2 AS e, interval '1 day' / 3 AS f, interval '1 hour' * 1.5 AS g;
SELECT (interval '1 day' > interval '2 hours')::int AS a, (interval '1 mon' = interval '30 days')::int AS b, (interval '25 hours' > '1 day')::int AS c;

-- case: date, timestamp and interval columns
CREATE TABLE ev (id int, d date, ts timestamp, iv interval);
INSERT INTO ev VALUES (1, '2024-01-31', '2024-01-31 10:00', '1 day'), (2, '2024-02-29', '2024-02-29 00:00:00', '2 hours'), (3, '2024-03-01', '2024-03-01 12:00:00', '1 mon');
SELECT id, d, ts, iv FROM ev ORDER BY id;
SELECT id, d + id AS a, ts + iv AS b, d - date '2024-01-01' AS c, ts - interval '1 hour' AS e FROM ev ORDER BY id;
SELECT id FROM ev ORDER BY iv;
SELECT id FROM ev WHERE d > '2024-02-01' AND ts < timestamp '2024-03-01 12:00' ORDER BY id;
CREATE VIEW ev_next AS SELECT id, d + 1 AS next, -iv AS neg FROM ev WHERE iv < interval '1 week';
SELECT * FROM ev_next ORDER BY id;

-- case: regular expression operators
CREATE TABLE names (n text);
INSERT INTO names VALUES ('pg_x'), ('abc'), ('Abd');
SELECT ('abc' ~ '^a')::int AS a, ('abc' ~ '^b')::int AS b, ('ABC' ~* '^a')::int AS c, ('abc' !~ 'z')::int AS d, ('abc' !~* 'A')::int AS e, ('pg_toast' ~ '^(pg_toast)$')::int AS f;
SELECT count(*) FROM names WHERE n !~ '^pg_' AND n ~* '^a';
SELECT n FROM names WHERE n ~ '^a';
