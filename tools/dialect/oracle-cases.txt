-- Differential cases for the Oracle dialect (xqt dialect-diff oracle).
-- Each case starts with a "-- case:" line and runs on an empty schema, in
-- Oracle Database (SQL*Plus, unaligned, with headers, dates formatted
-- 'YYYY-MM-DD HH24:MI:SS') and in `xqlite sql` with `.dialect oracle`; the
-- rows, column names and errors (by ORA- code) must be the same. Numbers
-- that print differently (.5 / 0.5, 2 / 2.0) and booleans (TRUE / 1) are
-- normalized; reals with many digits are avoided.

-- case: names fold to upper case, quoted ones keep theirs

INSERT INTO MIXED VALUES (1, 'a');
SELECT ID, "Camel", mixed.id FROM mixed;
SELECT table_name FROM user_tables;
