# views, CTEs, subqueries, joins, triggers with UPDATE OF, upsert, INSTEAD OF
CREATE TABLE p(id INTEGER PRIMARY KEY, name TEXT, qty INT);
CREATE TABLE q(pid INT REFERENCES p(id) ON DELETE CASCADE, note TEXT);
CREATE VIEW pv AS WITH c(x) AS (SELECT qty FROM p WHERE qty > 0) SELECT name, (SELECT count(*) FROM q WHERE q.pid = p.id) AS n FROM p WHERE qty IN (SELECT x FROM c) ORDER BY name;
CREATE VIEW pv2(a, b) AS SELECT p.name, q.note FROM p LEFT JOIN q ON p.id = q.pid;
CREATE VIEW pv3 AS SELECT * FROM p;
CREATE VIEW pv4 AS SELECT name AS nm FROM p GROUP BY nm HAVING count(*) > 0 ORDER BY 1;
CREATE VIEW pv5 AS SELECT p1.name, p2.qty FROM p AS p1, main.p AS p2 USING (id) WHERE EXISTS (SELECT 1 FROM q WHERE pid = p1.id);
CREATE TRIGGER pt AFTER DELETE ON p BEGIN DELETE FROM q WHERE pid = old.id; SELECT name FROM p WHERE qty = old.qty; END;
CREATE TRIGGER pt2 INSTEAD OF INSERT ON pv2 BEGIN INSERT INTO p(name) VALUES(new.a); END;
CREATE TRIGGER pt3 AFTER UPDATE OF qty, name ON p FOR EACH ROW WHEN old.qty IS NOT new.qty BEGIN INSERT INTO q(pid, note) SELECT new.id, new.name FROM p WHERE p.id = new.id; END;
CREATE TABLE u(k PRIMARY KEY, cnt);
CREATE TRIGGER pt4 AFTER INSERT ON p BEGIN INSERT INTO u(k, cnt) VALUES(new.name, 1) ON CONFLICT(k) DO UPDATE SET cnt = cnt + 1 WHERE excluded.k = u.k; END;
CREATE TRIGGER pt5 BEFORE UPDATE ON q BEGIN UPDATE p SET qty = qty + 1, name = new.note WHERE id = new.pid; END;
INSERT INTO p VALUES(1, 'apple', 3), (2, 'pear', 0);
INSERT INTO q VALUES(1, 'n1');
