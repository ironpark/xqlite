-- Differential cases of PostgreSQL views whose query SQLite has no
-- syntax for: LATERAL subqueries naming the FROM items before them
-- (stored as json_each() over their packed rows, viewlateral.zig), and
-- pg_get_viewdef() of views whose query is desugared or rewritten
-- (DISTINCT ON, GROUPING SETS, ROLLUP, CUBE, WITH TIES, LATERAL), which
-- prints the query as written (xqt dialect-diff pg
-- tools/dialect/pg-viewdef-cases.txt), in the format of pg-cases.txt:
-- each case starts with a "-- case:" line and runs on an empty
-- database, in PostgreSQL (psql, unaligned, with headers) and in
-- `xqlite sql` with `.dialect postgres`; the rows, column names and
-- error messages must be the same. The expected output is PostgreSQL
-- 18's.

-- case: LATERAL view: a column of the subquery by an item before it
CREATE TABLE e (id int, d int, s int);
INSERT INTO e VALUES (1, 1, 10), (2, 1, 30), (3, 2, 20), (4, 2, 20), (5, NULL, 5);
CREATE VIEW v AS SELECT a.id, b.n FROM e a, LATERAL (SELECT a.s + 1 AS n) b;
SELECT * FROM v ORDER BY id;
SELECT pg_typeof(n) FROM v LIMIT 1;

-- case: LEFT JOIN LATERAL ... ON true in a view: the next row of each group
CREATE TABLE e (id int, d int, s int);
INSERT INTO e VALUES (1, 1, 10), (2, 1, 30), (3, 2, 20), (4, 2, 20), (5, NULL, 5);
CREATE VIEW v AS SELECT a.id, b.id AS nid, b.big FROM e a LEFT JOIN LATERAL (SELECT id, s > 15 AS big FROM e b WHERE b.d = a.d AND b.id > a.id ORDER BY id LIMIT 1) b ON true;
SELECT * FROM v ORDER BY id;
SELECT id FROM v WHERE nid IS NULL ORDER BY id;

-- case: LEFT JOIN LATERAL with a condition on the subquery's column
CREATE TABLE e (id int, d int, s int);
INSERT INTO e VALUES (1, 1, 10), (2, 1, 30), (3, 2, 20), (4, 2, 20), (5, NULL, 5);
CREATE VIEW v AS SELECT a.id, x.m FROM e a LEFT JOIN LATERAL (SELECT max(s) AS m FROM e b WHERE b.d = a.d) x ON x.m > 20;
SELECT * FROM v ORDER BY id;

-- case: LATERAL view of several rows per row, with types kept
CREATE TABLE t (id int, n numeric(6,2), b boolean, x bytea, f float8, tx text);
INSERT INTO t VALUES (1, 1.5, true, decode('00ff', 'hex'), 0.1, 'é'), (2, NULL, false, NULL, -2.5e300, NULL);
CREATE VIEW v AS SELECT t.id, r.* FROM t, LATERAL (SELECT t.n * 2 AS n2, t.b AS bb, length(t.x) AS xx, t.f AS ff, t.tx || '!' AS tt UNION ALL SELECT t.n, NOT t.b, NULL, t.f / 10, NULL) r;
SELECT * FROM v ORDER BY id, n2;
SELECT pg_typeof(n2), pg_typeof(bb), pg_typeof(xx), pg_typeof(ff), pg_typeof(tt) FROM v LIMIT 1;

-- case: LATERAL view: SELECT * and an unqualified column next to json_each()'s names
CREATE TABLE e (id int, value int, key text);
INSERT INTO e VALUES (1, 10, 'a'), (2, 20, 'b');
CREATE VIEW v AS SELECT * FROM e, LATERAL (SELECT value * 2 AS v2, key || '!' AS k2) l;
SELECT * FROM v ORDER BY id;
SELECT id, value, v2 FROM v WHERE v2 > 20;

-- case: LATERAL view over a set-returning function and a later LATERAL
CREATE TABLE e (id int, d int);
INSERT INTO e VALUES (1, 2), (2, 1), (3, 0);
CREATE VIEW v AS SELECT e.id, g.k, h.sq FROM e, LATERAL (SELECT k FROM generate_series(1, e.d) k) g, LATERAL (SELECT g.k * g.k AS sq) h;
SELECT * FROM v ORDER BY id, k;

-- case: pg_get_viewdef() of a DISTINCT ON view
CREATE TABLE e (id int, d int, s int);
CREATE VIEW v AS SELECT DISTINCT ON (d) id, d FROM e ORDER BY d, id DESC;
SELECT pg_get_viewdef('v');
SELECT pg_get_viewdef('v', true);

-- case: pg_get_viewdef() of GROUPING SETS, ROLLUP and CUBE views
CREATE TABLE e (id int, d int, s int);
CREATE VIEW v1 AS SELECT d, s, count(*) FROM e GROUP BY GROUPING SETS ((d), (s), ());
CREATE VIEW v2 AS SELECT d, sum(s) FROM e GROUP BY ROLLUP (d);
CREATE VIEW v3 AS SELECT d, s, count(*) FROM e GROUP BY CUBE (d, s);
SELECT pg_get_viewdef('v1');
SELECT pg_get_viewdef('v2');
SELECT pg_get_viewdef('v3');

-- case: pg_get_viewdef() of a WITH TIES view
CREATE TABLE e (id int, d int, s int);
CREATE VIEW v1 AS SELECT id, d FROM e ORDER BY d FETCH FIRST 2 ROWS WITH TIES;
CREATE VIEW v2 AS SELECT id FROM e ORDER BY d OFFSET 1 FETCH FIRST 2 ROWS WITH TIES;
SELECT pg_get_viewdef('v1');
SELECT pg_get_viewdef('v2');
