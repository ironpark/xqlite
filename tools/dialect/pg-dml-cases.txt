-- Differential cases of PostgreSQL's DML beyond SQLite's: MERGE, the
-- data-modifying statements of WITH, ON CONFLICT ON CONSTRAINT,
-- RETURNING's OLD and NEW, OVERRIDING USER VALUE (xqt dialect-diff pg
-- tools/dialect/pg-dml-cases.txt), in the format of pg-cases.txt: each
-- case starts with a "-- case:" line and runs on an empty database, in
-- PostgreSQL (psql, unaligned, with headers) and in `xqlite sql` with
-- `.dialect postgres`; the rows, column names and error messages must
-- be the same. A statement's rows are what it returns (RETURNING) or a
-- query after it; a case of errors prints nothing else (psql's stderr
-- and stdout do not keep their order through podman). The expected
-- output is PostgreSQL 18's.

-- case: MERGE: every kind of WHEN clause, RETURNING merge_action()
CREATE TABLE t (id int PRIMARY KEY, a int, b boolean);
INSERT INTO t VALUES (1, 10, true), (2, 20, false), (3, 30, NULL);
MERGE INTO t USING (VALUES (1, 100), (4, 400), (5, 500)) AS s(id, a) ON t.id = s.id
WHEN MATCHED AND t.a > 50 THEN DELETE
WHEN MATCHED THEN UPDATE SET a = s.a + t.a
WHEN NOT MATCHED AND s.id = 5 THEN DO NOTHING
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a, DEFAULT)
WHEN NOT MATCHED BY SOURCE AND t.id = 3 THEN DELETE
WHEN NOT MATCHED BY SOURCE THEN UPDATE SET b = NOT b
RETURNING merge_action(), t.*, s.a AS sa;
SELECT * FROM t ORDER BY id;

-- case: MERGE without RETURNING, a table as the source, aliases
CREATE TABLE target (k int PRIMARY KEY, v text, n int DEFAULT 7);
CREATE TABLE src (k int, v text);
INSERT INTO target VALUES (1, 'one', 1), (2, 'two', 2);
INSERT INTO src VALUES (2, 'TWO'), (3, 'three'), (4, NULL);
MERGE INTO target tg USING src ON tg.k = src.k
WHEN MATCHED THEN UPDATE SET v = src.v, n = tg.n * 10
WHEN NOT MATCHED AND src.v IS NOT NULL THEN INSERT (k, v) VALUES (src.k, src.v);
SELECT * FROM target ORDER BY k;
MERGE INTO target AS x USING (SELECT k + 1 AS k, 'shifted' AS v FROM src) AS y ON x.k = y.k
WHEN MATCHED AND x.v = 'three' THEN DELETE
WHEN NOT MATCHED THEN INSERT VALUES (y.k, y.v);
SELECT * FROM target ORDER BY k;

-- case: MERGE: the first WHEN clause whose condition holds
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t SELECT g, g * 10 FROM generate_series(1, 6) g;
MERGE INTO t USING (SELECT g AS id FROM generate_series(1, 8) g) s ON t.id = s.id
WHEN MATCHED AND t.id % 3 = 0 THEN DELETE
WHEN MATCHED AND t.id % 2 = 0 THEN UPDATE SET a = -t.a
WHEN MATCHED THEN DO NOTHING
WHEN NOT MATCHED AND s.id = 7 THEN INSERT VALUES (s.id, 70)
WHEN NOT MATCHED THEN INSERT (id) VALUES (s.id);
SELECT * FROM t ORDER BY id;

-- case: MERGE: a row twice, DO NOTHING twice, a source row twice not matched
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t VALUES (1, 1);
MERGE INTO t USING (VALUES (1), (1)) AS s(id) ON t.id = s.id WHEN MATCHED THEN DO NOTHING;
MERGE INTO t USING (VALUES (1, 5), (1, 6), (2, 7)) AS s(id, a) ON t.id = s.id AND s.a = 5 WHEN MATCHED THEN UPDATE SET a = s.a;
SELECT * FROM t ORDER BY id;

-- case: MERGE: errors
CREATE TABLE t (id int PRIMARY KEY, a int CHECK (a < 100), b text NOT NULL DEFAULT 'b');
INSERT INTO t VALUES (1, 1, 'x'), (2, 2, 'y');
MERGE INTO t USING (VALUES (1), (1)) AS s(id) ON t.id = s.id WHEN MATCHED THEN UPDATE SET a = 0;
MERGE INTO t USING (VALUES (1), (1)) AS s(id) ON t.id = s.id WHEN MATCHED THEN DELETE;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN MATCHED THEN UPDATE SET id = 2;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN MATCHED THEN UPDATE SET a = 500;
MERGE INTO t USING (SELECT 3 AS id) s ON t.id = s.id WHEN NOT MATCHED THEN INSERT VALUES (s.id, 1, NULL);
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN MATCHED THEN UPDATE SET zz = 2;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN NOT MATCHED BY SOURCE THEN UPDATE SET a = s.id;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN NOT MATCHED THEN INSERT VALUES (t.id);
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN NOT MATCHED AND t.a = 1 THEN DO NOTHING;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN MATCHED THEN INSERT VALUES (1);
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN NOT MATCHED THEN DELETE;
MERGE INTO t USING (SELECT 1 AS id) s ON id = s.id WHEN MATCHED THEN DELETE;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN NOT MATCHED THEN INSERT VALUES (1, 2, 'c', 4);
MERGE INTO nope USING (SELECT 1 AS id) s ON nope.id = s.id WHEN MATCHED THEN DELETE;
MERGE INTO t USING nope ON t.id = nope.id WHEN MATCHED THEN DELETE;
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id;
SELECT merge_action();
MERGE INTO t USING (SELECT 1 AS id) s ON t.id = s.id WHEN MATCHED AND merge_action() = 'x' THEN DELETE;

-- case: MERGE: an error undoes the whole statement
CREATE TABLE t (id int PRIMARY KEY, a int CHECK (a < 100));
INSERT INTO t VALUES (1, 1), (2, 2);
MERGE INTO t USING (VALUES (1, 50), (2, 500), (3, 3)) AS s(id, a) ON t.id = s.id
WHEN MATCHED AND s.id = 1 THEN UPDATE SET a = s.a
WHEN MATCHED THEN UPDATE SET a = s.a
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a);
SELECT * FROM t ORDER BY id;

-- case: MERGE in a transaction: an error undoes the statement only
CREATE TABLE t (id int PRIMARY KEY, a int CHECK (a < 100));
BEGIN;
INSERT INTO t VALUES (1, 1);
SAVEPOINT s1;
MERGE INTO t USING (VALUES (1, 5), (2, 500)) AS s(id, a) ON t.id = s.id
WHEN MATCHED THEN UPDATE SET a = s.a
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a);
ROLLBACK TO s1;
MERGE INTO t USING (VALUES (1, 5), (2, 50)) AS s(id, a) ON t.id = s.id
WHEN MATCHED THEN UPDATE SET a = s.a
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a);
COMMIT;
SELECT * FROM t ORDER BY id;

-- case: MERGE: typed columns, defaults, identity and generated columns
CREATE TABLE t (id int GENERATED ALWAYS AS IDENTITY, k text UNIQUE, flag boolean DEFAULT true, amount numeric(8,2), note varchar(5), at date, doubled int GENERATED ALWAYS AS (length(k) * 2) STORED);
INSERT INTO t (k, amount) VALUES ('a', 1.5);
MERGE INTO t USING (VALUES ('a', 2.255, 'no', '2024-02-03'), ('bb', 3, 'yes', '2024-12-31')) AS s(k, amt, f, d) ON t.k = s.k
WHEN MATCHED THEN UPDATE SET amount = t.amount + s.amt::numeric, flag = s.f::boolean, at = s.d::date
WHEN NOT MATCHED THEN INSERT (k, amount, note, at) VALUES (s.k, s.amt::numeric, s.f, s.d::date)
RETURNING merge_action(), id, k, flag, amount, note, at, doubled;
SELECT * FROM t ORDER BY id;

-- case: MERGE: INSERT DEFAULT VALUES and OVERRIDING
CREATE TABLE t (id int GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, v text DEFAULT 'dflt');
MERGE INTO t USING (VALUES (10), (20)) AS s(n) ON false WHEN NOT MATCHED THEN INSERT DEFAULT VALUES;
MERGE INTO t USING (VALUES (10, 'x')) AS s(n, v) ON false WHEN NOT MATCHED THEN INSERT OVERRIDING USER VALUE VALUES (s.n, s.v);
MERGE INTO t USING (VALUES (10, 'y')) AS s(n, v) ON false WHEN NOT MATCHED THEN INSERT (id, v) VALUES (s.n, DEFAULT);
SELECT * FROM t ORDER BY id;

-- case: MERGE: RETURNING * is the source's columns, then the target's
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t VALUES (1, 1);
MERGE INTO t USING (SELECT 1 AS id, 2 AS b UNION ALL SELECT 2, 3) s ON t.id = s.id
WHEN MATCHED THEN UPDATE SET a = s.b
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.b)
RETURNING *, merge_action(), pg_typeof(merge_action());

-- case: MERGE: RETURNING old and new
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t VALUES (1, 10), (2, 20);
MERGE INTO t USING (VALUES (1, 5), (3, 30)) AS s(id, a) ON t.id = s.id
WHEN MATCHED THEN UPDATE SET a = s.a
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a)
WHEN NOT MATCHED BY SOURCE THEN DELETE
RETURNING merge_action(), old.a AS old_a, new.a AS new_a, old.*, t.id;
MERGE INTO t USING (VALUES (1)) AS s(id) ON t.id = s.id WHEN MATCHED THEN DELETE
RETURNING WITH (OLD AS o, NEW AS n) o.*, n.id, t.a, o.a + 1 AS next;

-- case: WITH ... MERGE, MERGE in WITH
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t VALUES (1, 1), (2, 2);
WITH s AS (SELECT 1 AS id, 99 AS a UNION ALL SELECT 5, 55) MERGE INTO t USING s ON t.id = s.id
WHEN MATCHED THEN UPDATE SET a = s.a WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.a) RETURNING t.*;
WITH m AS (MERGE INTO t USING (SELECT 7 AS id) s ON t.id = s.id WHEN NOT MATCHED THEN INSERT VALUES (s.id, 70) RETURNING merge_action() AS act, t.*) SELECT * FROM m;
WITH m AS (MERGE INTO t USING (VALUES (1), (2), (9)) s(id) ON t.id = s.id WHEN MATCHED THEN DELETE RETURNING t.id) SELECT count(*), sum(id) FROM m;
SELECT * FROM t ORDER BY id;

-- case: MERGE of a schema's table, a parameterless prepared statement
CREATE SCHEMA app;
CREATE TABLE app.items (sku text PRIMARY KEY, qty int NOT NULL);
INSERT INTO app.items VALUES ('a', 1);
MERGE INTO app.items i USING (VALUES ('a', 5), ('b', 2)) AS v(sku, qty) ON i.sku = v.sku
WHEN MATCHED THEN UPDATE SET qty = i.qty + v.qty
WHEN NOT MATCHED THEN INSERT VALUES (v.sku, v.qty);
SELECT * FROM app.items ORDER BY sku;

-- case: WITH: DELETE ... RETURNING into another table
CREATE TABLE t (id int PRIMARY KEY, a int, b text);
INSERT INTO t VALUES (1, 10, 'x'), (2, 20, 'y'), (3, 30, 'z');
CREATE TABLE log (id int, a int, what text);
WITH d AS (DELETE FROM t WHERE id = 3 RETURNING *) INSERT INTO log SELECT id, a, 'deleted' FROM d;
SELECT * FROM log;
SELECT * FROM t ORDER BY id;

-- case: WITH: UPDATE, INSERT and the statement's query
CREATE TABLE t (id int PRIMARY KEY, a int, b text);
INSERT INTO t VALUES (1, 10, 'x'), (2, 20, 'y');
CREATE TABLE log (id int, a int, what text);
WITH u AS (UPDATE t SET a = a + 1 RETURNING id, a AS na), i AS (INSERT INTO log SELECT id, na, 'updated' FROM u RETURNING *) SELECT * FROM i ORDER BY id;
WITH x AS (SELECT 5 AS id), i AS (INSERT INTO t SELECT id, 50, 'w' FROM x RETURNING id * 2 AS twice, b) SELECT * FROM i;
WITH d AS (DELETE FROM t WHERE id = 5 RETURNING b) UPDATE t SET b = (SELECT b FROM d) WHERE id = 1 RETURNING *;
WITH a AS (INSERT INTO log VALUES (9, 9, 'a')), b AS (INSERT INTO log VALUES (10, 10, 'b') RETURNING id) SELECT count(*) FROM b;
SELECT * FROM log ORDER BY id;
WITH d AS (DELETE FROM log RETURNING id) SELECT sum(id) FROM d;
WITH d (k) AS (DELETE FROM t WHERE id = 2 RETURNING id) SELECT k FROM d;
SELECT * FROM t ORDER BY id;

-- case: WITH: a recursive query before, the RETURNING's old and new
CREATE TABLE t (id int PRIMARY KEY, b text);
INSERT INTO t VALUES (1, 'x'), (3, 'z');
CREATE TABLE log (id int);
WITH RECURSIVE r(n) AS (SELECT 1 UNION ALL SELECT n + 1 FROM r WHERE n < 3), i AS (INSERT INTO log SELECT n FROM r RETURNING id) SELECT * FROM i;
WITH u AS (UPDATE t SET b = b || '!' RETURNING id, old.b AS ob, new.b AS nb) SELECT * FROM u ORDER BY id;
WITH d AS (DELETE FROM t WHERE id = 3 RETURNING old.*, new.b AS gone) SELECT * FROM d;

-- case: WITH: errors
CREATE TABLE t (id int PRIMARY KEY, a int);
INSERT INTO t VALUES (1, 1);
WITH d AS (DELETE FROM t) SELECT * FROM d;
SELECT * FROM (WITH d AS (DELETE FROM t RETURNING *) SELECT * FROM d) x;
WITH d AS (UPDATE t SET a = 1/0 RETURNING *) SELECT * FROM d;
WITH d AS (DELETE FROM t RETURNING zz) SELECT * FROM d;
WITH d AS (INSERT INTO t VALUES (1, 2) RETURNING id) SELECT * FROM d;

-- case: WITH: an error undoes every statement
CREATE TABLE t (id int PRIMARY KEY, a int);
CREATE TABLE log (id int PRIMARY KEY);
INSERT INTO t VALUES (1, 1), (2, 2);
WITH d AS (DELETE FROM t RETURNING id) INSERT INTO log SELECT 1 FROM d;
SELECT (SELECT count(*) FROM t) AS t, (SELECT count(*) FROM log) AS log;

-- case: ON CONFLICT ON CONSTRAINT
CREATE TABLE t (id int PRIMARY KEY, a int, b int, CONSTRAINT ab UNIQUE (a, b));
CREATE TABLE u (k text CONSTRAINT u_key PRIMARY KEY, v int);
INSERT INTO t VALUES (1, 1, 1);
INSERT INTO t VALUES (1, 5, 5) ON CONFLICT ON CONSTRAINT t_pkey DO UPDATE SET a = excluded.a RETURNING *;
INSERT INTO t VALUES (2, 5, 5) ON CONFLICT ON CONSTRAINT ab DO NOTHING RETURNING *;
INSERT INTO t VALUES (3, 5, 5) ON CONFLICT ON CONSTRAINT ab DO UPDATE SET id = excluded.id RETURNING *;
INSERT INTO u VALUES ('a', 1);
INSERT INTO u VALUES ('a', 2) ON CONFLICT ON CONSTRAINT u_key DO UPDATE SET v = u.v + excluded.v RETURNING *;

-- case: ON CONFLICT ON CONSTRAINT: errors
CREATE TABLE t (id int PRIMARY KEY, a int);
CREATE UNIQUE INDEX ua ON t (a);
INSERT INTO t VALUES (1, 1) ON CONFLICT ON CONSTRAINT nope DO NOTHING;
INSERT INTO t VALUES (2, 9) ON CONFLICT ON CONSTRAINT ua DO NOTHING;

-- case: RETURNING old and new
CREATE TABLE t (id int PRIMARY KEY, a int, b text);
INSERT INTO t VALUES (1, 10, 'x'), (2, 20, 'y');
INSERT INTO t VALUES (3, 30, 'z') RETURNING old.a, new.a, old.*, new.*;
INSERT INTO t VALUES (3, 31, 'z') ON CONFLICT (id) DO UPDATE SET a = excluded.a RETURNING old.a AS o, new.a AS n;
UPDATE t SET a = a * 2 WHERE id < 3 RETURNING id, old.a, new.a, a, t.a, new.a - old.a AS diff;
UPDATE t SET b = 'q' WHERE id = 1 RETURNING WITH (OLD AS o, NEW AS n) o.b, n.b, o.a = n.a AS same;
DELETE FROM t WHERE id = 2 RETURNING old.*, new.b;

-- case: RETURNING old and new: errors
CREATE TABLE t (id int PRIMARY KEY, a int);
UPDATE t SET a = 0 RETURNING old.zz;
UPDATE t SET a = 0 RETURNING new.zz;

-- case: OVERRIDING USER VALUE, UPDATE of a column the table has not
CREATE TABLE u (id int GENERATED ALWAYS AS IDENTITY, v text);
INSERT INTO u OVERRIDING USER VALUE VALUES (10, 'a');
INSERT INTO u OVERRIDING USER VALUE SELECT 20, 'b';
INSERT INTO u (id, v) OVERRIDING USER VALUE VALUES (30, 'c'), (DEFAULT, 'd');
INSERT INTO u (v, id) OVERRIDING USER VALUE SELECT 'e', 50;
SELECT * FROM u ORDER BY id;

-- case: UPDATE of a column the table has not
CREATE TABLE t (id int, a int);
UPDATE t SET zz = 1;
UPDATE t AS x SET (a, zz) = (1, 2);

-- case: UPDATE ... SET (a, b) = (SELECT ...), DELETE ... USING
CREATE TABLE t (id int PRIMARY KEY, a int, b text);
CREATE TABLE s (id int, a int, b text);
INSERT INTO t VALUES (1, 1, 'a'), (2, 2, 'b');
INSERT INTO s VALUES (1, 10, 'ten'), (2, 20, 'twenty');
UPDATE t SET (a, b) = (SELECT s.a, s.b FROM s WHERE s.id = t.id) WHERE id = 1 RETURNING *;
DELETE FROM t USING s WHERE s.id = t.id AND s.a = 20 RETURNING t.*;
SELECT * FROM t ORDER BY id;

-- case: MERGE: jsonb, uuid, interval and array columns, the source the target itself
CREATE TABLE t (id int PRIMARY KEY, j jsonb, u uuid, iv interval, tags text[]);
INSERT INTO t VALUES (1, '{"a": 1}', 'a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11', '1 day', '{x}');
MERGE INTO t USING (SELECT id + 1 AS id, j || '{"b": 2}' AS j, u, iv * 2 AS iv, tags || 'y'::text AS tags FROM t) s ON t.id = s.id
WHEN MATCHED THEN UPDATE SET j = s.j
WHEN NOT MATCHED THEN INSERT VALUES (s.id, s.j, s.u, s.iv, s.tags)
RETURNING t.*, s.j -> 'b' AS b, pg_typeof(s.iv);
MERGE INTO t t2 USING t ON t2.id = t.id + 1 WHEN MATCHED THEN UPDATE SET j = t.j, tags = t2.tags || t.tags RETURNING t2.id, t2.j, t2.tags;
SELECT * FROM t ORDER BY id;

-- case: UPDATE ... FROM a table of text columns
CREATE TABLE target (k int PRIMARY KEY, v text, n int, c varchar(3), b boolean);
CREATE TABLE src (k int, w text, flag boolean);
INSERT INTO target VALUES (1, 'one', 1, 'a', NULL), (2, 'two', 2, 'b', NULL);
INSERT INTO src VALUES (2, 'TWO', true), (3, 'three', false);
UPDATE target AS tg SET v = src.w, n = 5, c = src.w, b = src.flag FROM src WHERE src.k = tg.k RETURNING tg.*;
UPDATE target SET v = src.flag, c = 'x' || src.k FROM src WHERE src.k = target.k RETURNING target.*;

-- case: the DEFAULT of an integer primary key
CREATE TABLE pk (id int PRIMARY KEY DEFAULT 99, v text DEFAULT 'x');
INSERT INTO pk DEFAULT VALUES;

INSERT INTO pk (v) VALUES ('dup') ON CONFLICT (id) DO UPDATE SET v = 'upd' RETURNING *;
INSERT INTO pk (v) VALUES ('n') ON CONFLICT DO NOTHING;
CREATE TABLE pk2 (id bigint PRIMARY KEY DEFAULT 7, v text);

INSERT INTO pk3 (v) VALUES ('big');
