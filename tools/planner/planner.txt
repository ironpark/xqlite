# basic single-table access paths
CREATE TABLE t1(a INTEGER PRIMARY KEY, b, c, d TEXT);
CREATE INDEX t1b ON t1(b);
CREATE INDEX t1cd ON t1(c, d);
CREATE UNIQUE INDEX t1d ON t1(d);
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<300)
INSERT INTO t1 SELECT i, i%17, i%5, 'v'||i FROM n;

SELECT * FROM t1 WHERE a=5
SELECT * FROM t1 WHERE a>10 AND a<20

SELECT * FROM t1 WHERE b>15

SELECT * FROM t1 WHERE b IN (1,2,3)

# joins
CREATE TABLE p(id INTEGER PRIMARY KEY, name TEXT);
CREATE TABLE q(id INTEGER PRIMARY KEY, pid INTEGER, v);
CREATE INDEX qpid ON q(pid);
CREATE TABLE r(x, y);
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<100)
INSERT INTO p SELECT i, 'n'||i FROM n;
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<400)
INSERT INTO q SELECT i, i%120, i*3 FROM n;
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<50)
INSERT INTO r SELECT i%13, i FROM n;
==
SELECT * FROM p, q WHERE p.id=q.pid AND p.id<5
SELECT * FROM p JOIN q ON p.id=q.pid WHERE q.v>1100
SELECT p.name, q.v FROM q, p WHERE q.pid=p.id ORDER BY q.v LIMIT 10
SELECT * FROM p LEFT JOIN q ON q.pid=p.id WHERE p.id>95
SELECT * FROM p LEFT JOIN q ON q.pid=p.id WHERE q.id IS NULL
SELECT * FROM r, p WHERE r.x=p.id
SELECT * FROM r JOIN q ON r.x=q.v
SELECT * FROM p RIGHT JOIN r ON r.x=p.id
SELECT * FROM p FULL JOIN r ON r.x=p.id
SELECT * FROM r FULL JOIN q ON r.y=q.pid WHERE q.id<20 OR q.id IS NULL
SELECT * FROM p, q, r WHERE p.id=q.pid AND q.id=r.y
SELECT count(*) FROM p, q WHERE p.id=q.pid
SELECT * FROM p WHERE EXISTS (SELECT 1 FROM q WHERE q.pid=p.id AND q.v>1000)
SELECT * FROM p WHERE id NOT IN (SELECT pid FROM q)
SELECT * FROM p CROSS JOIN q WHERE q.pid=p.id AND p.id=3
SELECT * FROM (SELECT pid, count(*) c FROM q GROUP BY pid) s JOIN p ON p.id=s.pid WHERE s.c>3
SELECT * FROM r a, r b WHERE a.x=b.y
SELECT * FROM q WHERE pid IN (SELECT id FROM p WHERE name LIKE 'n1%')

# without rowid, partial and expression indexes
CREATE TABLE w(k TEXT PRIMARY KEY, v INT, z) WITHOUT ROWID;
CREATE INDEX wv ON w(v);
CREATE TABLE e(a, b, c);
CREATE INDEX epart ON e(b) WHERE c IS NOT NULL;
CREATE INDEX eexpr ON e(a+b);
CREATE INDEX elower ON e(lower(c));
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<200)
INSERT INTO w SELECT 'k'||i, i%9, i FROM n;
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<200)
INSERT INTO e SELECT i, i%7, CASE WHEN i%3 THEN 'C'||i END FROM n;
==
SELECT * FROM w WHERE k='k5'
SELECT * FROM w WHERE k>'k9'
SELECT * FROM w WHERE v=3
SELECT k FROM w WHERE v=3 ORDER BY k
SELECT * FROM w ORDER BY k LIMIT 4
SELECT * FROM e WHERE b=2 AND c IS NOT NULL
SELECT * FROM e WHERE b=2
SELECT * FROM e WHERE a+b=10
SELECT * FROM e WHERE lower(c)='c10'
SELECT a+b FROM e ORDER BY a+b LIMIT 3

# DML
CREATE TABLE u(a INTEGER PRIMARY KEY, b, c);
CREATE INDEX ub ON u(b);
WITH RECURSIVE n(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM n WHERE i<100)
INSERT INTO u SELECT i, i%10, i FROM n;
==
UPDATE u SET c=0 WHERE a=5
UPDATE u SET c=c+1 WHERE b=3
UPDATE u SET b=b+1 WHERE b=3
DELETE FROM u WHERE a=9
DELETE FROM u WHERE b=4
DELETE FROM u WHERE a>90
UPDATE u SET c=1 WHERE a IN (1,2,3)
