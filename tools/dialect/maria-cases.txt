-- Differential cases for the MariaDB dialect (xqt dialect-diff mariadb).
-- Each case starts with a "-- case:" line and runs on an empty database
-- called main, in MariaDB (the mariadb client, batch mode) and in
-- `xqlite sql` with `.dialect mariadb`; the rows, column names and error
-- messages (the "ERROR:" line) must be the same. What the two print
-- differently by design (docs/mariadb.md: DECIMAL as REAL, the
-- engine's own error messages, NOW()) is avoided; numbers with a
-- fraction are compared without trailing zeros.

-- case: arithmetic and operators
SELECT 1 + 1, 'a' 'b', 7 / 2, 7 DIV 2, -7 DIV 2, 7 % 3, 7 MOD 3;
SELECT 1 || 0, 0 || 0, 1 && 0, !0, NOT 1, 1 XOR 1, 1 XOR 0, NULL XOR 1;
SELECT NULL <=> NULL, 1 <=> NULL, 1 = NULL, 5 ^ 3, 1 << 3, 6 & 3, 6 | 1;
SELECT NOT 1 = 2, !1 = 2, 2 + 3 * 4, (2 + 3) * 4, 10 - 2 - 3, 2 * 3 DIV 4;
SELECT 10/4, 1/0, 5 DIV 0, '7' / 2, 7 / '2';

-- case: comparisons and predicates
SELECT 'abc' = 'ABC', 'abc' LIKE 'A%', BINARY 'abc' = 'ABC', 'a' = 'A' COLLATE utf8mb4_bin;
SELECT 2 BETWEEN 1 AND 3, 5 NOT BETWEEN 1 AND 3, 2 IN (1,2), 2 NOT IN (1,3), NULL IS NULL, 1 IS TRUE, 0 IS NOT FALSE;
SELECT 'a%' LIKE 'a\%', 'ab' LIKE 'a\%', 'a_b' LIKE 'a\_b', 'axb' LIKE 'a\_b', 'a%' LIKE 'a|%' ESCAPE '|';
SELECT CASE WHEN 1 THEN 'y' ELSE 'n' END, CASE 2 WHEN 1 THEN 'a' WHEN 2 THEN 'b' END, 'a' IN ('A', 'b');

-- case: literals
SELECT TRUE, FALSE, X'41', 0x41 = X'41', b'101' + 0, 0b11 + 0, 1e3, .5;
SELECT 1 /*! + 1 */ /*!999999 + 100 */ /*M!100100 + 10 */ AS x, 3--1;
SELECT 9223372036854775807 + 0, -9223372036854775807 - 1;

-- case: string functions
SELECT IFNULL(NULL, 2), IF(1 > 2, 'y', 'n'), CONCAT('a', 1, 'b'), CONCAT('a', NULL), CONCAT_WS('-', 'a', NULL, 'b');
SELECT LENGTH('abc'), CHAR_LENGTH('abc'), UCASE('a'), LCASE('B'), SUBSTRING('hello', 2, 3), SUBSTRING('hello' FROM 2), MID('hello', 2, 1);
SELECT TRIM('  x  '), TRIM(LEADING 'x' FROM 'xxaxx'), TRIM(TRAILING FROM 'a  '), LOCATE('l', 'hello'), LOCATE('l', 'hello', 4), POSITION('l' IN 'hello');
SELECT LPAD('5', 3, '0'), RPAD('ab', 4, '.'), LEFT('hello', 2), RIGHT('hello', 3), REPEAT('ab', 2), REVERSE('abc'), HEX(255), HEX('A');
SELECT FORMAT(1234567.891, 2), FIND_IN_SET('b', 'a,b,c'), FIELD('b', 'a', 'b'), ELT(2, 'a', 'b');
SELECT REPLACE('aXbX', 'X', '-'), INSTR('hello', 'l'), ASCII('A'), CHAR(65, 66), SPACE(2) = '  ', INSERT('abcd', 2, 1, 'X');

-- case: numeric functions
SELECT FLOOR(1.5), CEIL(1.2), ROUND(2.5), ROUND(-2.5), ROUND(2.567, 2), TRUNCATE(1.29, 1), ABS(-3);
SELECT GREATEST(1, 5, 3), LEAST(4, 2), MOD(10, 3), SIGN(-4), SQRT(16), POW(2, 3);
SELECT CAST('12abc' AS SIGNED), CAST(1.5 AS CHAR), CAST('3.456' AS DECIMAL(5,2)), CAST('2024-01-02 10:00:00' AS DATE), CONVERT('abc', CHAR(2));
SELECT CAST(3.7 AS DECIMAL), CAST(-3.7 AS DECIMAL(5,0)), CAST(2.5 AS DECIMAL(4,1)), CAST(7 AS DOUBLE) / 2;

-- case: date functions
SELECT DATE_ADD('2024-01-31', INTERVAL 1 MONTH), '2024-01-01' + INTERVAL 1 DAY, DATE_SUB('2024-03-01 00:00:00', INTERVAL 1 SECOND);
SELECT YEAR('2024-05-06'), MONTH('2024-05-06'), DAY('2024-05-06'), DAYOFWEEK('2024-05-06'), DATEDIFF('2024-03-01', '2024-02-01'), LAST_DAY('2023-02-10');
SELECT DATE_FORMAT('2024-05-06 13:04:05', '%Y/%m/%d %H:%i:%s %W %M %D'), EXTRACT(YEAR FROM '2024-05-06');
SELECT TIMESTAMPDIFF(MONTH, '2024-01-15', '2024-03-14'), TIMESTAMPDIFF(DAY, '2024-01-01', '2024-01-31'), TIMESTAMPADD(DAY, 2, '2024-02-28');
SELECT HOUR('10:20:30'), MINUTE('2024-01-01 10:20:30'), SECOND('10:20:30'), DAYNAME('2024-05-06'), MONTHNAME('2024-05-06');
SELECT TIME('2024-01-01 10:20:30'), DATE('2024-01-01 10:20:30'), STR_TO_DATE('01/02/2024', '%m/%d/%Y');
SELECT ISNULL(NULL), COALESCE(NULL, 1), NULLIF(1, 1), CAST('2024-02-30' AS DATE);
