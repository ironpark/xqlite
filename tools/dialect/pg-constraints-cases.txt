-- Differential cases of SET CONSTRAINTS: the mode of each DEFERRABLE
-- foreign key named, or of all, until the transaction ends, and the
-- check of the violations deferred so far when it makes them IMMEDIATE
-- (xqt dialect-diff pg tools/dialect/pg-constraints-cases.txt), in the
-- format of pg-cases.txt: each case starts with a "-- case:" line and
-- runs on an empty database, in PostgreSQL (psql, unaligned, with
-- headers) and in `xqlite sql` with `.dialect postgres`; the rows,
-- column names and error messages must be the same. A case of errors
-- prints nothing else (psql's stderr and stdout do not keep their order
-- through podman). The expected output is PostgreSQL 18's.

-- case: SET CONSTRAINTS: what pg_constraint says of the declarations
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (x int REFERENCES p DEFERRABLE INITIALLY DEFERRED, y int, z int REFERENCES p NOT DEFERRABLE,
  w int REFERENCES p INITIALLY DEFERRED, v int,
  CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE INITIALLY IMMEDIATE);
ALTER TABLE c ADD CONSTRAINT cv FOREIGN KEY (v) REFERENCES p DEFERRABLE;
SELECT conname, condeferrable, condeferred FROM pg_constraint WHERE conrelid = 'c'::regclass ORDER BY conname;
SELECT pg_get_constraintdef(oid) FROM pg_constraint WHERE conname IN ('cy', 'cv') ORDER BY conname;

-- case: SET CONSTRAINTS: a name defers its foreign key, not the others
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (y int, z int REFERENCES p DEFERRABLE, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS cy DEFERRED;
INSERT INTO c VALUES (1, NULL);
INSERT INTO p VALUES (1);
COMMIT;
BEGIN;
SET CONSTRAINTS public.cy DEFERRED;
INSERT INTO c VALUES (2, NULL);
DELETE FROM c WHERE y = 2;
COMMIT;
SELECT y, z FROM c ORDER BY y;

-- case: SET CONSTRAINTS: a foreign key not named stays immediate
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (y int, z int REFERENCES p DEFERRABLE, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS cy DEFERRED;
INSERT INTO c VALUES (NULL, 7);
ROLLBACK;

-- case: SET CONSTRAINTS: the mode lasts until the transaction ends
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (y int, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS cy DEFERRED;
COMMIT;
BEGIN;
INSERT INTO c VALUES (9);
ROLLBACK;

-- case: SET CONSTRAINTS: ALL, then a name IMMEDIATE
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (x int, y int, CONSTRAINT cx FOREIGN KEY (x) REFERENCES p DEFERRABLE, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS ALL DEFERRED;
SET CONSTRAINTS cy IMMEDIATE;
INSERT INTO c VALUES (8, NULL);
INSERT INTO c VALUES (NULL, 8);
ROLLBACK;

-- case: SET CONSTRAINTS: ALL defers the deferrable ones only
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (x int REFERENCES p DEFERRABLE, z int REFERENCES p);
BEGIN;
SET CONSTRAINTS ALL DEFERRED;
INSERT INTO c VALUES (3, NULL);
INSERT INTO c VALUES (NULL, 3);
ROLLBACK;

-- case: SET CONSTRAINTS: IMMEDIATE checks the violations deferred so far
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (x int REFERENCES p DEFERRABLE INITIALLY DEFERRED, y int, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
INSERT INTO p VALUES (1);
BEGIN;
INSERT INTO c VALUES (5, NULL);
SET CONSTRAINTS cy IMMEDIATE;
SET CONSTRAINTS c_x_fkey IMMEDIATE;
ROLLBACK;
BEGIN;
INSERT INTO c VALUES (1, NULL), (6, NULL);
SET CONSTRAINTS ALL IMMEDIATE;
ROLLBACK;
BEGIN;
SET CONSTRAINTS ALL DEFERRED;
INSERT INTO c VALUES (NULL, 7), (NULL, 1);
SET CONSTRAINTS c_x_fkey IMMEDIATE;
SET CONSTRAINTS cy IMMEDIATE;
ROLLBACK;

-- case: SET CONSTRAINTS: IMMEDIATE after the violations were fixed
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (x int REFERENCES p DEFERRABLE INITIALLY DEFERRED);
BEGIN;
INSERT INTO c VALUES (5);
INSERT INTO p VALUES (5);
SET CONSTRAINTS ALL IMMEDIATE;
INSERT INTO c VALUES (5);
COMMIT;
SELECT x FROM c;

-- case: SET CONSTRAINTS: the names must be deferrable constraints
CREATE TABLE p (id int PRIMARY KEY, n int NOT NULL CHECK (n > 0));
CREATE TABLE c (y int, z int REFERENCES p, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS nosuch DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS c_z_fkey DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS cy, c_z_fkey DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS p_pkey DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS p_n_check DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS nosch.cy DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS pg_catalog.cy IMMEDIATE;
ROLLBACK;
BEGIN;
SET CONSTRAINTS nosuch IMMEDIATE;
ROLLBACK;

-- case: SET CONSTRAINTS: IMMEDIATE of a constraint that is not deferrable
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (z int REFERENCES p);
BEGIN;
SET CONSTRAINTS c_z_fkey, p_pkey IMMEDIATE;
INSERT INTO p VALUES (1);
INSERT INTO c VALUES (1);
COMMIT;
SELECT z FROM c;

-- case: SET CONSTRAINTS: outside a transaction block
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (y int, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
SET CONSTRAINTS cy DEFERRED;
SET CONSTRAINTS ALL IMMEDIATE;

-- case: SET CONSTRAINTS: a savepoint's modes
CREATE TABLE p (id int PRIMARY KEY);
CREATE TABLE c (y int, CONSTRAINT cy FOREIGN KEY (y) REFERENCES p DEFERRABLE);
BEGIN;
SET CONSTRAINTS ALL DEFERRED;
SAVEPOINT a;
SET CONSTRAINTS ALL IMMEDIATE;
ROLLBACK TO a;
INSERT INTO c VALUES (1);
ROLLBACK;
BEGIN;
SAVEPOINT a;
SET CONSTRAINTS cy DEFERRED;
RELEASE a;
INSERT INTO c VALUES (2);
DELETE FROM c;
COMMIT;
SELECT count(*) FROM c;

-- case: SET CONSTRAINTS: a schema's foreign key
CREATE SCHEMA s;
CREATE TABLE s.p (id int PRIMARY KEY);
CREATE TABLE s.c (y int REFERENCES s.p DEFERRABLE);
BEGIN;
SET CONSTRAINTS s.c_y_fkey DEFERRED;
INSERT INTO s.c VALUES (4);
INSERT INTO s.p VALUES (4);
COMMIT;
SELECT y FROM s.c;

-- case: SET CONSTRAINTS: a schema's foreign key not on the search_path
CREATE SCHEMA s;
CREATE TABLE s.p (id int PRIMARY KEY);
CREATE TABLE s.c (y int REFERENCES s.p DEFERRABLE);
BEGIN;
SET CONSTRAINTS c_y_fkey DEFERRED;
ROLLBACK;
BEGIN;
SET CONSTRAINTS s.c_y_fkey DEFERRED;
INSERT INTO s.c VALUES (4);
SET CONSTRAINTS s.c_y_fkey IMMEDIATE;
ROLLBACK;
