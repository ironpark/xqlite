# journal and locking modes of in-memory (':memory:') and temporary ('')
# databases, attached to a rollback-journal database.

CREATE INDEX tb ON t(b);
WITH RECURSIVE c(i) AS (SELECT 1 UNION ALL SELECT i+1 FROM c WHERE i<300) INSERT INTO t SELECT i, 'v' || i FROM c;
.each ATTACH ':memory:' AS m

# their modes.

.each ATTACH ':memory:' AS m
.each ATTACH '' AS e
